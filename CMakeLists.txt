cmake_minimum_required(VERSION 3.22)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MicroApp VERSION 0.0.2 LANGUAGES C)
endif()

add_library(microapp_sdk INTERFACE)
add_library(MicroApp::sdk ALIAS microapp_sdk)
target_include_directories(microapp_sdk INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

add_library(microapp STATIC src/microapp.c src/microapp_crc32.c src/microapp_native.c src/microapp_store.c)
add_library(MicroApp::core ALIAS microapp)
target_link_libraries(microapp PUBLIC MicroApp::sdk)
target_compile_features(microapp PUBLIC c_std_11)
set_target_properties(microapp PROPERTIES VERSION 0.0.2)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]|cortex" OR CMAKE_C_COMPILER MATCHES "arm-none-eabi")
    enable_language(ASM)
    target_sources(microapp PRIVATE src/microapp_arm_call.S)
endif()
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/MicroAppArm.cmake)

option(MICROAPP_BUILD_TESTS "Build standalone MicroApp host tests" OFF)
if(MICROAPP_BUILD_TESTS)
    enable_testing()
    add_executable(microapp_tests tests/test_microapp.c)
    target_link_libraries(microapp_tests PRIVATE MicroApp::core)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(microapp PRIVATE -Wall -Wextra -Werror -Wpedantic)
        target_compile_options(microapp_tests PRIVATE -Wall -Wextra -Werror -Wpedantic)
    endif()
    add_test(NAME microapp_core COMMAND microapp_tests)
    add_executable(microapp_native_tests tests/test_microapp_native.c)
    target_link_libraries(microapp_native_tests PRIVATE MicroApp::core)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(microapp_native_tests PRIVATE -Wall -Wextra -Werror -Wpedantic)
    endif()
    add_test(NAME microapp_native COMMAND microapp_native_tests)
    add_executable(microapp_store_tests tests/test_microapp_store.c)
    target_link_libraries(microapp_store_tests PRIVATE MicroApp::core)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(microapp_store_tests PRIVATE -Wall -Wextra -Werror -Wpedantic)
    endif()
    add_test(NAME microapp_store COMMAND microapp_store_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME microapp_native_tools
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_microapp_tools.py)
    endif()
endif()
